#
# Installation (Python 3.11 — 3.12 n'a pas ensurepip sur cette machine) :
#   cd backend
#   python3.11 -m venv .venv-timecopilot
#   ./.venv-timecopilot/bin/pip install -r requirements-timecopilot.txt
#
# Le backend n'importe jamais ce venv : il l'appelle en sous-processus via
# app/services/timecopilot_runner.py, ce qui garantit l'étanchéité des deux
